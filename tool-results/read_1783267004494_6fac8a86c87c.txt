     1→"use client";
     2→
     3→import * as React from "react";
     4→import Link from "next/link";
     5→import { useRouter } from "next/navigation";
     6→import {
     7→  MapPin,
     8→  Truck,
     9→  Tag,
    10→  CreditCard,
    11→  Check,
    12→  ChevronLeft,
    13→  Wallet,
    14→  Plus,
    15→  Loader2,
    16→  AlertCircle,
    17→  ShoppingBag,
    18→} from "lucide-react";
    19→import { toast } from "sonner";
    20→
    21→import { Button } from "@/components/ui/button";
    22→import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
    23→import { Input } from "@/components/ui/input";
    24→import { Label } from "@/components/ui/label";
    25→import { RadioGroup, RadioGroupItem } from "@/components/ui/radio-group";
    26→import { Separator } from "@/components/ui/separator";
    27→import { Skeleton } from "@/components/ui/skeleton";
    28→import { AuthGuard } from "@/components/common/auth-guard";
    29→import { Breadcrumb } from "@/components/common/breadcrumb";
    30→import { EmptyState } from "@/components/common/empty-state";
    31→import { AddressFormDialog } from "@/features/account/components/address-form-dialog";
    32→import {
    33→  useAddresses,
    34→  useWallet,
    35→} from "@/features/account/hooks";
    36→import { useCart } from "@/features/cart/hooks";
    37→import {
    38→  useCreateOrder,
    39→  useApplyDiscountCode,
    40→} from "@/features/checkout/hooks";
    41→import { shippingCompaniesService, paymentGatewaysService } from "@/services";
    42→import type {
    43→  Address,
    44→  DiscountApplyResult,
    45→  PaymentMethod,
    46→  ShippingCompany,
    47→  PaymentGateway,
    48→} from "@/types/domain";
    49→import { formatToman, formatPrice, toPersianDigits } from "@/utils/format";
    50→import { cn } from "@/lib/utils";
    51→
    52→type Step = "address" | "shipping" | "review" | "payment";
    53→
    54→const STEPS: { id: Step; label: string; icon: React.ReactNode }[] = [
    55→  { id: "address", label: "آدرس", icon: <MapPin className="size-4" /> },
    56→  { id: "shipping", label: "ارسال", icon: <Truck className="size-4" /> },
    57→  { id: "review", label: "بررسی", icon: <Tag className="size-4" /> },
    58→  { id: "payment", label: "پرداخت", icon: <CreditCard className="size-4" /> },
    59→];
    60→
    61→export default function CheckoutPage() {
    62→  return (
    63→    <AuthGuard redirectTo="/login">
    64→      <CheckoutContent />
    65→    </AuthGuard>
    66→  );
    67→}
    68→
    69→function CheckoutContent() {
    70→  const router = useRouter();
    71→  const { data: cart, isLoading: cartLoading } = useCart();
    72→  const { data: addresses, isLoading: addressesLoading } = useAddresses();
    73→  const { data: wallet } = useWallet();
    74→
    75→  const [step, setStep] = React.useState<Step>("address");
    76→  const [selectedAddressId, setSelectedAddressId] = React.useState<string | null>(null);
    77→  const [selectedShippingId, setSelectedShippingId] = React.useState<string | null>(null);
    78→  const [discountCode, setDiscountCode] = React.useState("");
    79→  const [discountResult, setDiscountResult] = React.useState<DiscountApplyResult | null>(null);
    80→  const [paymentMethod, setPaymentMethod] = React.useState<PaymentMethod>("GATEWAY");
    81→  const [addressDialogOpen, setAddressDialogOpen] = React.useState(false);
    82→
    83→  // Shipping companies + payment gateways (one-time fetch).
    84→  const [shippingCompanies, setShippingCompanies] = React.useState<ShippingCompany[]>([]);
    85→  const [gateways, setGateways] = React.useState<PaymentGateway[]>([]);
    86→  React.useEffect(() => {
    87→    shippingCompaniesService.list().then(setShippingCompanies).catch(() => {});
    88→    paymentGatewaysService.list().then(setGateways).catch(() => {});
    89→  }, []);
    90→
    91→  // Auto-select first address + first shipping company.
    92→  React.useEffect(() => {
    93→    if (addresses && addresses.length > 0 && !selectedAddressId) {
    94→      const def = addresses.find((a) => a.isDefault) ?? addresses[0];
    95→      setSelectedAddressId(def.id);
    96→    }
    97→  }, [addresses, selectedAddressId]);
    98→
    99→  React.useEffect(() => {
   100→    if (shippingCompanies.length > 0 && !selectedShippingId) {
   101→      setSelectedShippingId(shippingCompanies[0].id);
   102→    }
   103→  }, [shippingCompanies, selectedShippingId]);
   104→
   105→  // Redirect to cart if empty.
   106→  React.useEffect(() => {
   107→    if (!cartLoading && cart && cart.items.length === 0) {
   108→      router.replace("/cart");
   109→    }
   110→  }, [cartLoading, cart, router]);
   111→
   112→  // Calculate totals (safe even if cart is empty).
   113→  const cartTotal = cart?.total ?? 0;
   114→  const selectedShipping = shippingCompanies.find((s) => s.id === selectedShippingId);
   115→  const shippingCost = selectedShipping?.baseCost ?? 0;
   116→  const discountAmount = discountResult?.discountAmount ?? 0;
   117→  const payableTotal = Math.max(0, cartTotal - discountAmount + shippingCost);
   118→  const walletBalance = wallet?.balance ?? 0;
   119→  const canPayWithWallet = walletBalance >= payableTotal && payableTotal > 0;
   120→
   121→  const applyDiscount = useApplyDiscountCode();
   122→  const createOrder = useCreateOrder();
   123→
   124→  // Auto-switch payment method if wallet can't cover.
   125→  React.useEffect(() => {
   126→    if (paymentMethod === "WALLET" && !canPayWithWallet && payableTotal > 0) {
   127→      setPaymentMethod("GATEWAY");
   128→    }
   129→  }, [paymentMethod, canPayWithWallet, payableTotal]);
   130→
   131→  if (cartLoading || addressesLoading) {
   132→    return <CheckoutSkeleton />;
   133→  }
   134→
   135→  if (!cart || cart.items.length === 0) {
   136→    return (
   137→      <div className="container-site py-12">
   138→        <EmptyState
   139→          icon={<ShoppingBag className="size-16" />}
   140→          title="سبد خرید شما خالی است"
   141→          description="برای ادامه خرید، ابتدا محصولاتی را به سبد اضافه کنید."
   142→          action={
   143→            <Button asChild>
   144→              <Link href="/products">مشاهده محصولات</Link>
   145→            </Button>
   146→          }
   147→        />
   148→      </div>
   149→    );
   150→  }
   151→
   152→  const selectedAddress = addresses?.find((a) => a.id === selectedAddressId);
   153→
   154→  const onApplyDiscount = () => {
   155→    if (!discountCode.trim()) return;
   156→    applyDiscount.mutate(discountCode, {
   157→      onSuccess: (data) => {
   158→        setDiscountResult(data);
   159→        setDiscountCode("");
   160→      },
   161→      onError: () => {
   162→        setDiscountResult(null);
   163→      },
   164→    });
   165→  };
   166→
   167→  const onPlaceOrder = () => {
   168→    if (!selectedAddressId || !selectedShippingId) {
   169→      toast.error("آدرس و روش ارسال را انتخاب کنید");
   170→      return;
   171→    }
   172→    const gatewaySlug = paymentMethod !== "WALLET" ? gateways[0]?.slug : undefined;
   173→    if (paymentMethod !== "WALLET" && !gatewaySlug) {
   174→      toast.error("درگاه پرداخت در دسترس نیست");
   175→      return;
   176→    }
   177→    createOrder.mutate({
   178→      addressId: selectedAddressId,
   179→      shippingCompanyId: selectedShippingId,
   180→      paymentMethod,
   181→      gatewaySlug,
   182→      discountCode: discountResult?.code,
   183→    });
   184→  };
   185→
   186→  const currentStepIndex = STEPS.findIndex((s) => s.id === step);
   187→
   188→  return (
   189→    <div className="container-site py-6">
   190→      <Breadcrumb
   191→        items={[
   192→          { name: "خانه", url: "/" },
   193→          { name: "سبد خرید", url: "/cart" },
   194→          { name: "تسویه حساب", url: "/checkout" },
   195→        ]}
   196→      />
   197→
   198→      <h1 className="mb-6 text-xl font-bold text-foreground sm:text-2xl">تسویه حساب</h1>
   199→
   200→      {/* Stepper */}
   201→      <div className="mb-6 flex items-center justify-between gap-2 overflow-x-auto pb-2">
   202→        {STEPS.map((s, i) => {
   203→          const isActive = s.id === step;
   204→          const isDone = i < currentStepIndex;
   205→          return (
   206→            <React.Fragment key={s.id}>
   207→              <button
   208→                onClick={() => {
   209→                  if (i <= currentStepIndex) setStep(s.id);
   210→                }}
   211→                className={cn(
   212→                  "flex items-center gap-2 rounded-full px-3 py-1.5 text-sm font-medium transition-colors",
   213→                  isActive
   214→                    ? "bg-primary text-primary-foreground"
   215→                    : isDone
   216→                      ? "bg-primary/10 text-primary"
   217→                      : "bg-muted text-muted-foreground",
   218→                )}
   219→              >
   220→                <span className="flex size-5 items-center justify-center rounded-full bg-white/20 text-xs">
   221→                  {isDone ? <Check className="size-3" /> : toPersianDigits(i + 1)}
   222→                </span>
   223→                <span className="whitespace-nowrap">{s.label}</span>
   224→              </button>
   225→              {i < STEPS.length - 1 && (
   226→                <div className={cn("h-px flex-1", isDone ? "bg-primary" : "bg-border")} />
   227→              )}
   228→            </React.Fragment>
   229→          );
   230→        })}
   231→      </div>
   232→
   233→      <div className="grid grid-cols-1 gap-6 lg:grid-cols-3">
   234→        <div className="space-y-4 lg:col-span-2">
   235→          {/* Step: Address */}
   236→          {step === "address" && (
   237→            <Card>
   238→              <CardHeader>
   239→                <CardTitle className="flex items-center gap-2 text-base">
   240→                  <MapPin className="size-5 text-primary" />
   241→                  انتخاب آدرس تحویل
   242→                </CardTitle>
   243→              </CardHeader>
   244→              <CardContent className="space-y-3">
   245→                {addresses && addresses.length === 0 ? (
   246→                  <div className="rounded-lg border border-dashed border-border p-6 text-center">
   247→                    <p className="mb-3 text-sm text-muted-foreground">
   248→                      هنوز آدرسی ثبت نکرده‌اید
   249→                    </p>
   250→                    <Button onClick={() => setAddressDialogOpen(true)}>
   251→                      <Plus className="size-4" />
   252→                      افزودن آدرس جدید
   253→                    </Button>
   254→                  </div>
   255→                ) : (
   256→                  <>
   257→                    {addresses?.map((addr) => (
   258→                      <AddressRadioCard
   259→                        key={addr.id}
   260→                        address={addr}
   261→                        selected={addr.id === selectedAddressId}
   262→                        onSelect={() => setSelectedAddressId(addr.id)}
   263→                        onEdit={() => setAddressDialogOpen(true)}
   264→                      />
   265→                    ))}
   266→                    <Button
   267→                      variant="outline"
   268→                      size="sm"
   269→                      onClick={() => setAddressDialogOpen(true)}
   270→                    >
   271→                      <Plus className="size-4" />
   272→                      افزودن آدرس جدید
   273→                    </Button>
   274→                  </>
   275→                )}
   276→              </CardContent>
   277→            </Card>
   278→          )}
   279→
   280→          {/* Step: Shipping */}
   281→          {step === "shipping" && (
   282→            <Card>
   283→              <CardHeader>
   284→                <CardTitle className="flex items-center gap-2 text-base">
   285→                  <Truck className="size-5 text-primary" />
   286→                  انتخاب روش ارسال
   287→                </CardTitle>
   288→              </CardHeader>
   289→              <CardContent className="space-y-3">
   290→                {shippingCompanies.length === 0 ? (
   291→                  <p className="text-sm text-muted-foreground">
   292→                    در حال حاضر روش ارسالی موجود نیست.
   293→                  </p>
   294→                ) : (
   295→                  shippingCompanies.map((sc) => (
   296→                    <ShippingRadioCard
   297→                      key={sc.id}
   298→                      company={sc}
   299→                      selected={sc.id === selectedShippingId}
   300→                      onSelect={() => setSelectedShippingId(sc.id)}
   301→                    />
   302→                  ))
   303→                )}
   304→              </CardContent>
   305→            </Card>
   306→          )}
   307→
   308→          {/* Step: Review */}
   309→          {step === "review" && (
   310→            <Card>
   311→              <CardHeader>
   312→                <CardTitle className="flex items-center gap-2 text-base">
   313→                  <Tag className="size-5 text-primary" />
   314→                  مرور سفارش و کد تخفیف
   315→                </CardTitle>
   316→              </CardHeader>
   317→              <CardContent className="space-y-4">
   318→                {/* Discount code */}
   319→                <div className="space-y-2">
   320→                  <Label className="text-sm font-medium">کد تخفیف</Label>
   321→                  {discountResult ? (
   322→                    <div className="flex items-center justify-between rounded-lg bg-success/10 px-3 py-2 text-sm">
   323→                      <div>
   324→                        <span className="font-medium text-success">
   325→                          کد «{discountResult.code}» اعمال شد
   326→                        </span>
   327→                        <span className="mr-2 text-muted-foreground nums-fa">
   328→                          ({toPersianDigits(discountResult.discountAmount)} تومان تخفیف)
   329→                        </span>
   330→                      </div>
   331→                      <Button
   332→                        variant="ghost"
   333→                        size="sm"
   334→                        className="h-7 text-destructive"
   335→                        onClick={() => {
   336→                          setDiscountResult(null);
   337→                        }}
   338→                      >
   339→                        حذف
   340→                      </Button>
   341→                    </div>
   342→                  ) : (
   343→                    <div className="flex gap-2">
   344→                      <Input
   345→                        type="text"
   346→                        placeholder="مثال: WELCOME20"
   347→                        value={discountCode}
   348→                        onChange={(e) => setDiscountCode(e.target.value.toUpperCase())}
   349→                        dir="ltr"
   350→                        className="text-left"
   351→                      />
   352→                      <Button
   353→                        variant="outline"
   354→                        size="sm"
   355→                        onClick={onApplyDiscount}
   356→                        disabled={applyDiscount.isPending || !discountCode.trim()}
   357→                      >
   358→                        {applyDiscount.isPending && <Loader2 className="size-4 animate-spin" />}
   359→                        اعمال
   360→                      </Button>
   361→                    </div>
   362→                  )}
   363→                </div>
   364→
   365→                <Separator />
   366→
   367→                {/* Cart items review */}
   368→                <div className="space-y-3">
   369→                  <h4 className="text-sm font-medium">اقلام سفارش</h4>
   370→                  {cart.items.map((item) => (
   371→                    <div
   372→                      key={item.id}
   373→                      className="flex items-center justify-between gap-3 text-sm"
   374→                    >
   375→                      <div className="flex items-center gap-2">
   376→                        <span className="flex size-6 items-center justify-center rounded-md bg-muted text-xs nums-fa">
   377→                          {toPersianDigits(item.quantity)}
   378→                        </span>
   379→                        <div>
   380→                          <p className="font-medium text-foreground">{item.productName}</p>
   381→                          {item.attributesLabel && (
   382→                            <p className="text-xs text-muted-foreground">
   383→                              {item.attributesLabel}
   384→                            </p>
   385→                          )}
   386→                        </div>
   387→                      </div>
   388→                      <span className="font-medium nums-fa">
   389→                        {formatPrice(item.lineTotal)}
   390→                      </span>
   391→                    </div>
   392→                  ))}
   393→                </div>
   394→
   395→                <Separator />
   396→
   397→                {/* Address + shipping summary */}
   398→                <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   399→                  <div className="rounded-lg bg-muted/30 p-3 text-sm">
   400→                    <p className="mb-1 flex items-center gap-1 font-medium text-foreground">
   401→                      <MapPin className="size-3.5 text-primary" />
   402→                      آدرس تحویل
   403→                    </p>
   404→                    {selectedAddress ? (
   405→                      <>
   406→                        <p className="text-muted-foreground">{selectedAddress.receiverName}</p>
   407→                        <p className="text-muted-foreground">
   408→                          {selectedAddress.province}، {selectedAddress.city}
   409→                        </p>
   410→                        <p className="text-muted-foreground">{selectedAddress.fullAddress}</p>
   411→                      </>
   412→                    ) : (
   413→                      <p className="text-destructive">آدرسی انتخاب نشده</p>
   414→                    )}
   415→                  </div>
   416→                  <div className="rounded-lg bg-muted/30 p-3 text-sm">
   417→                    <p className="mb-1 flex items-center gap-1 font-medium text-foreground">
   418→                      <Truck className="size-3.5 text-primary" />
   419→                      روش ارسال
   420→                    </p>
   421→                    {selectedShipping ? (
   422→                      <>
   423→                        <p className="text-muted-foreground">{selectedShipping.name}</p>
   424→                        {selectedShipping.estimatedDaysMin && (
   425→                          <p className="text-muted-foreground">
   426→                            تحویل: {toPersianDigits(selectedShipping.estimatedDaysMin)}
   427→                            {selectedShipping.estimatedDaysMax
   428→                              ? ` تا ${toPersianDigits(selectedShipping.estimatedDaysMax)}`
   429→                              : ""}{" "}
   430→                            روز
   431→                          </p>
   432→                        )}
   433→                      </>
   434→                    ) : (
   435→                      <p className="text-destructive">روش ارسالی انتخاب نشده</p>
   436→                    )}
   437→                  </div>
   438→                </div>
   439→              </CardContent>
   440→            </Card>
   441→          )}
   442→
   443→          {/* Step: Payment */}
   444→          {step === "payment" && (
   445→            <Card>
   446→              <CardHeader>
   447→                <CardTitle className="flex items-center gap-2 text-base">
   448→                  <CreditCard className="size-5 text-primary" />
   449→                  روش پرداخت
   450→                </CardTitle>
   451→              </CardHeader>
   452→              <CardContent className="space-y-3">
   453→                <RadioGroup
   454→                  value={paymentMethod}
   455→                  onValueChange={(v) => setPaymentMethod(v as PaymentMethod)}
   456→                >
   457→                  <PaymentOptionCard
   458→                    value="GATEWAY"
   459→                    selected={paymentMethod === "GATEWAY"}
   460→                    icon={<CreditCard className="size-5" />}
   461→                    title="پرداخت آنلاین (درگاه)"
   462→                    description={`مبلغ ${formatPrice(payableTotal)} تومان از درگاه بانکی پرداخت می‌شود`}
   463→                  />
   464→                  <PaymentOptionCard
   465→                    value="WALLET"
   466→                    selected={paymentMethod === "WALLET"}
   467→                    icon={<Wallet className="size-5" />}
   468→                    title="پرداخت از کیف پول"
   469→                    description={
   470→                      canPayWithWallet
   471→                        ? `موجودی شما: ${formatPrice(walletBalance)} تومان`
   472→                        : `موجودی ناکافی (${formatPrice(walletBalance)} تومان)`
   473→                    }
   474→                    disabled={!canPayWithWallet}
   475→                  />
   476→                  <PaymentOptionCard
   477→                    value="MIXED"
   478→                    selected={paymentMethod === "MIXED"}
   479→                    icon={<CreditCard className="size-5" />}
   480→                    title="پرداخت ترکیبی (کیف پول + درگاه)"
   481→                    description={
   482→                      canPayWithWallet
   483→                        ? "کل مبلغ از کیف پول کسر می‌شود"
   484→                        : `${formatPrice(walletBalance)} تومان از کیف پول، ${formatPrice(
   485→                            payableTotal - walletBalance,
   486→                          )} تومان از درگاه`
   487→                    }
   488→                    disabled={walletBalance === 0}
   489→                  />
   490→                </RadioGroup>
   491→
   492→                <div className="rounded-lg bg-info/10 p-3 text-xs text-info">
   493→                  <AlertCircle className="mb-1 size-4" />
   494→                  {paymentMethod === "WALLET" && "سفارش بلافاصله پس از پرداخت پردازش می‌شود."}
   495→                  {paymentMethod === "GATEWAY" &&
   496→                    "پس از ثبت سفارش، به درگاه پرداخت منتقل می‌شوید."}
   497→                  {paymentMethod === "MIXED" &&
   498→                    "مبلغ کیف پول بلافاصله کسر می‌شود؛ بقیه از درگاه پرداخت می‌شود."}
   499→                </div>
   500→              </CardContent>
   501→            </Card>
   502→          )}
   503→
   504→          {/* Navigation buttons */}
   505→          <div className="flex items-center justify-between">
   506→            {currentStepIndex > 0 ? (
   507→              <Button
   508→                variant="ghost"
   509→                onClick={() => setStep(STEPS[currentStepIndex - 1].id)}
   510→              >
   511→                <ChevronLeft className="size-4 rotate-180" />
   512→                مرحله قبل
   513→              </Button>
   514→            ) : (
   515→              <Button asChild variant="ghost">
   516→                <Link href="/cart">
   517→                  <ChevronLeft className="size-4 rotate-180" />
   518→                  بازگشت به سبد
   519→                </Link>
   520→              </Button>
   521→            )}
   522→
   523→            {currentStepIndex < STEPS.length - 1 ? (
   524→              <Button
   525→                onClick={() => setStep(STEPS[currentStepIndex + 1].id)}
   526→                disabled={
   527→                  (step === "address" && !selectedAddressId) ||
   528→                  (step === "shipping" && !selectedShippingId)
   529→                }
   530→              >
   531→                مرحله بعد
   532→                <ChevronLeft className="size-4" />
   533→              </Button>
   534→            ) : (
   535→              <Button
   536→                onClick={onPlaceOrder}
   537→                disabled={createOrder.isPending}
   538→                size="lg"
   539→              >
   540→                {createOrder.isPending && <Loader2 className="size-4 animate-spin" />}
   541→                ثبت سفارش
   542→                <Check className="size-4" />
   543→              </Button>
   544→            )}
   545→          </div>
   546→        </div>
   547→
   548→        {/* Order Summary sidebar */}
   549→        <div className="lg:col-span-1">
   550→          <Card className="sticky top-32 border-border/60">
   551→            <CardHeader>
   552→              <CardTitle className="text-base">خلاصه سفارش</CardTitle>
   553→            </CardHeader>
   554→            <CardContent className="space-y-3 text-sm">
   555→              <div className="flex justify-between">
