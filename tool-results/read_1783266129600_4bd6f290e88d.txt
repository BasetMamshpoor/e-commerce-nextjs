     1→/**
     2→ * Domain entity types — REWRITTEN to match updated API.md (July 2025).
     3→ *
     4→ * KEY CHANGES:
     5→ *   - All IDs are now `number` (integer auto-increment), NOT string (cuid).
     6→ *   - Products use `basePrice` + `priceAdjustment` model (not per-variant price).
     7→ *   - Discount is at product level, not variant level.
     8→ *   - View tracking is automatic (no POST /:id/view).
     9→ *   - GET /by-id/:id added for frontend ID-based routes.
    10→ *   - Image management via PUT /:id (multipart), no separate image endpoints.
    11→ *   - Comparison: only GET /?productIds=1,2,3 — no add/remove/clear.
    12→ *   - Media: inline uploads with entityType, usage tracking, download.
    13→ *   - New: Stories, Newsletter, Search, Landing Page, Admin Notifications, Withdrawals.
    14→ *   - Comments: authorId/authorName, isLiked, commentableType, blog post comments.
    15→ *   - Orders: CUSTOMER-only checkout, trackingCode/packageNumber, detailed returns.
    16→ *   - Attributes: isDisplay field added.
    17→ */

   881→export function getProductCategories(
   882→  product: Pick<Product, "categories">,
   883→): Category[] {
   884→  if (!product.categories) return [];
   885→  if (product.categories.length === 0) return [];
   886→  const first = product.categories[0] as unknown;
   887→  if (first && typeof first === "object" && "category" in first) {
   888→    return (product.categories as ProductCategoryLink[]).map((c) => c.category);
   889→  }
   890→  return product.categories as Category[];
   891→}
